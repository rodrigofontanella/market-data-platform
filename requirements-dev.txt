# Development and test dependencies.
#
# Deliberately NOT installed into the production images -- docker/Dockerfile.python
# copies only the per-app requirements.txt files.
#
# Setup, from the repository root:
#
#   python3 -m venv .venv
#   .venv/bin/pip install -r requirements-dev.txt
#   .venv/bin/pip install -r apps/consumer/requirements.txt
#   .venv/bin/pip install -r apps/api/requirements.txt
#   .venv/bin/pip install -e libs/market_core
#   .venv/bin/pip install -e libs/market_testing
#
# The consumer's runtime requirements are needed even though the unit tests
# touch neither Kafka nor PostgreSQL: app/consumer.py imports confluent_kafka at
# module level, and app/database.py calls create_engine() at import time, which
# eagerly imports the psycopg driver.
#
# Everything here is pinned to an exact version. These are the versions the test
# suite has actually been run against; a floating range means a green suite today
# and an unexplained failure tomorrow, from a dependency nobody changed on
# purpose. Upgrade deliberately: bump the pin, run ./scripts/check.sh --all,
# commit the pin and the result together.
#
# This is not a lockfile. It pins what we import plus the transitive packages
# that materially affect behaviour (SQLAlchemy, docker); the rest of the
# dependency tree still floats. If that becomes a problem, the answer is
# pip-tools or uv, not more lines here.

pytest==9.1.1

# Starlette's TestClient (apps/api/tests/integration) is built on httpx2, not
# httpx. Nothing in the services imports it, so it belongs here rather than in
# apps/api/requirements.txt.
#
# It was installed by hand and never declared. `pip show httpx2` reports an
# empty Required-by, and the production image has starlette without it -- so
# nothing pulled it in, here or anywhere. The api suite collected on this
# machine and would not have collected in any venv built from the sequence
# above. What surfaced it: `pip show httpx` finding nothing while `import
# httpx` also failed, because Starlette 1.x wants httpx2 and the name in the
# docs was right all along.
httpx2==2.12.0

# Integration tests (apps/consumer and apps/api) only. These need a
# reachable Docker daemon -- testcontainers starts a real postgres:16 per test
# session. They are excluded by default; ./scripts/check.sh --all includes them.
#
# psycopg carries the [binary] extra on purpose: the pure-Python package needs
# libpq and a compiler at install time, which the binary wheel avoids.
alembic==1.20.0
docker==7.2.0
psycopg[binary]==3.3.5
SQLAlchemy==2.0.52
testcontainers==4.15.0
ruff==0.16.7
